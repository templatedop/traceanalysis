GETTING STARTED
=============================================================================

{{- if .Values.server.enabled }}

1. Get the application URL:
{{- if .Values.server.ingress.enabled }}
{{- range $host := .Values.server.ingress.hosts }}
  http{{ if $.Values.server.ingress.tls }}s{{ end }}://{{ $host.host }}{{ (first $host.paths).path }}
{{- end }}
{{- else if contains "NodePort" .Values.server.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ include "observability-analysis.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "observability-analysis.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ include "observability-analysis.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.server.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  You can watch the status with:
    kubectl get --namespace {{ include "observability-analysis.namespace" . }} svc -w {{ include "observability-analysis.fullname" . }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ include "observability-analysis.namespace" . }} {{ include "observability-analysis.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.server.service.port }}
{{- else if contains "ClusterIP" .Values.server.service.type }}
  kubectl --namespace {{ include "observability-analysis.namespace" . }} port-forward svc/{{ include "observability-analysis.fullname" . }} 8080:{{ .Values.server.service.port }}
  echo "Visit http://127.0.0.1:8080 to access the API"
{{- end }}

2. Check deployment status:
  kubectl --namespace {{ include "observability-analysis.namespace" . }} get pods -l "app.kubernetes.io/name={{ include "observability-analysis.name" . }}"

3. View logs:
  kubectl --namespace {{ include "observability-analysis.namespace" . }} logs -l "app.kubernetes.io/name={{ include "observability-analysis.name" . }}" -f

{{- end }}
